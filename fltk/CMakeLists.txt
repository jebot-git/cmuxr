cmake_minimum_required(VERSION 3.21)
project(cmuxr-fltk LANGUAGES CXX)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig zlib)
add_executable(cmuxr-fltk main.cpp)
target_compile_features(cmuxr-fltk PRIVATE cxx_std_17)
target_compile_options(cmuxr-fltk PRIVATE -Wall -Wextra)
target_link_libraries(cmuxr-fltk PRIVATE fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::FONTCONFIG)
include(GNUInstallDirs)
install(TARGETS cmuxr-fltk RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

include(desktop-menu.cmake)
de_xr_desktop_menu(cmuxr-fltk)
